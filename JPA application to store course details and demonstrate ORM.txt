mapping.
Technology: JPA Entity + EJB service
Where to write: `src/java/Course.java` and `src/java/CourseBean.java`; configure `AppPU`
Easy explanation: ORM maps the Course Java object to a database table.
Code
// File: Course.java
import javax.persistence.*;
@Entity
public class Course {
    @Id @GeneratedValue(strategy=GenerationType.AUTO) private Long id;
    private String courseName;
    private int durationMonths;
    public Course() { }
    public Course(String n,int d) { courseName=n; durationMonths=d; }
    public Long getId(){return id;}
    public String getCourseName(){return courseName;}
    public int getDurationMonths(){return durationMonths;}
}

import java.util.*;
import javax.ejb.Stateless;
import javax.persistence.*;
@Stateless
public class CourseBean {
    @PersistenceContext(unitName="AppPU") private EntityManager em;
    public void save(String n,int d){ em.persist(new Course(n,d)); }
    public List<Course> all(){ return em.createQuery("SELECT c FROM Course c",Course.class).getResultList(); }
}
